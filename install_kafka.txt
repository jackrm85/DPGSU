4.4 Download e verifica del pacchetto
cd /tmp
DOWNLOAD_DIR=$(mktemp -d /tmp/kafka-install.XXXXXX)
cd "$DOWNLOAD_DIR"
KAFKA_VERSION=4.3.1
SCALA_VERSION=2.13
BASE="kafka_${SCALA_VERSION}-${KAFKA_VERSION}"
curl -fLO "https://downloads.apache.org/kafka/${KAFKA_VERSION}/${BASE}.tgz"
curl -fLO "https://downloads.apache.org/kafka/${KAFKA_VERSION}/${BASE}.tgz.sha512"
sha512sum "${BASE}.tgz"
cat "${BASE}.tgz.sha512"
# Confrontare l'hash calcolato con quello pubblicato prima di proseguire.
# Eseguire solo dopo corrispondenza SHA-512 confermata; il controllo non sovrascrive nulla.
if [ -e /opt/kafka ] || [ -L /opt/kafka ] || [ -e "/opt/${BASE}" ] || [ -L "/opt/${BASE}" ]; then
  echo "STOP: installazione o collegamento Kafka già presente; applicare il piano di ag-giornamento."
else
  tar -xzf "${BASE}.tgz" -C /opt && ln -sT "/opt/${BASE}" /opt/kafka
fi
# Proseguire soltanto se estrazione e creazione del collegamento sono riuscite.
chown -R root:kafka "/opt/${BASE}"
find "/opt/${BASE}" -type d -exec chmod 0750 {} \;
find "/opt/${BASE}" -type f -exec chmod 0640 {} \;
chmod 0750 /opt/${BASE}/bin/*.sh




4.5 Configurazione KRaft standalone
cp /opt/kafka/config/server.properties /opt/kafka/config/server.properties.orig
tee /opt/kafka/config/server.properties >/dev/null <<'EOF'
process.roles=broker,controller
node.id=1
controller.quorum.bootstrap.servers=localhost:9093
controller.listener.names=CONTROLLER
listeners=PLAINTEXT://0.0.0.0:9092,CONTROLLER://127.0.0.1:9093
advertised.listeners=PLAINTEXT://psndsctkfka021:9092
listener.security.protocol.map=CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT
inter.broker.listener.name=PLAINTEXT
log.dirs=/data/kafka
num.network.threads=3
num.io.threads=8
socket.send.buffer.bytes=102400
socket.receive.buffer.bytes=102400
socket.request.max.bytes=104857600
num.partitions=3
num.recovery.threads.per.data.dir=1
offsets.topic.replication.factor=1
transaction.state.log.replication.factor=1
transaction.state.log.min.isr=1
log.retention.hours=168
log.segment.bytes=1073741824
log.retention.check.interval.ms=300000
auto.create.topics.enable=false
delete.topic.enable=true
EOF
chown root:kafka /opt/kafka/config/server.properties
chmod 0640 /opt/kafka/config/server.properties



4.6 Inizializzazione storage KRaft
find /data/kafka -mindepth 1 -maxdepth 1 -print
# Output atteso vuoto; altrimenti STOP, non eseguire format.
sudo -u kafka test -w /data/kafka
sudo -u kafka test -w /var/log/kafka
grep -nE 'LOG_DIR|kafka.logs.dir|exec ' /opt/kafka/bin/kafka-run-class.sh
grep -nE 'kafka.logs.dir|fileName|filePattern' /opt/kafka/config/log4j2.yaml
KAFKA_CLUSTER_ID=$(sudo -u kafka env JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64 LOG_DIR=/var/log/kafka /opt/kafka/bin/kafka-storage.sh random-uuid)
echo "Cluster ID: ${KAFKA_CLUSTER_ID}"
sudo -u kafka env JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64 LOG_DIR=/var/log/kafka /opt/kafka/bin/kafka-storage.sh format --standalone \
  --cluster-id "${KAFKA_CLUSTER_ID}" \
  --config /opt/kafka/config/server.properties
sudo -u kafka env JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64 LOG_DIR=/var/log/kafka /opt/kafka/bin/kafka-storage.sh info \
  --config /opt/kafka/config/server.properties
