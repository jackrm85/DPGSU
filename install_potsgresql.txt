5.3 Installazione repository PGDG e PostgreSQL 15
apt install -y postgresql-common
/usr/share/postgresql-common/pgdg/apt.postgresql.org.sh
apt update
apt install -y postgresql-15 postgresql-client-15 postgresql-contrib-15
pg_lsclusters
/usr/lib/postgresql/15/bin/psql --version
sudo -u postgres /usr/lib/postgresql/15/bin/psql -p 5432 -c 'SHOW server_version;'


5.4 Hardening iniziale e ascolto rete
cp /etc/postgresql/15/main/postgresql.conf /etc/postgresql/15/main/postgresql.conf.$(date +%F_%H%M).bak
cp /etc/postgresql/15/main/pg_hba.conf /etc/postgresql/15/main/pg_hba.conf.$(date +%F_%H%M).bak
su - postgres
psql -c "ALTER SYSTEM SET password_encryption = 'scram-sha-256';"
psql -c "ALTER SYSTEM SET listen_addresses = '<IP_POSTGRES>,localhost';"
psql -c "ALTER SYSTEM SET log_connections = 'on';"
psql -c "ALTER SYSTEM SET log_disconnections = 'on';"
psql -c "ALTER SYSTEM SET log_line_prefix = '%m [%p] %q%u@%d ';"
# Sostituire <IP_APP>/32 con IP o subnet reale
printf '
host all all <IP_APP>/32 scram-sha-256
' | sudo tee -a /etc/postgresql/15/main/pg_hba.conf

# Impostare Logging
logging_collector = on
log_destination = 'stderr'
log_directory = 'log'
log_filename = 'postgresql-%Y-%m-%d_%H%M%S.log'
log_rotation_age = 1d
log_rotation_size = 200MB
log_min_duration_statement = 1000
log_checkpoints = on
log_lock_waits = on
log_line_prefix = '%m [%p] %u@%d %r '

# Impostare eventuali parametri di tuning
shared_buffers
effective_cache_size
work_mem
maintenance_work_mem
max_connections

systemctl restart postgresql@15-main
systemctl status postgresql@15-main --no-pager


5.5 Creazione database e utenza applicativa
# Generare la password nel vault; non inserirla nello storico shell.
su – postgres
psql <<'SQL'
CREATE ROLE app_owner LOGIN;
CREATE DATABASE appdb OWNER app_owner ENCODING 'UTF8' TEMPLATE template0;
REVOKE CREATE ON SCHEMA public FROM PUBLIC;
\c appdb
ALTER SCHEMA public OWNER TO app_owner;
SQL
# Impostazione password in modo interattivo
psql -c "\password app_owner"
# Verifiche
psql -c "\du+ app_owner"
psql -c "\l+ appdb"
