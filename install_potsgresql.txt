5.3 Installazione repository PGDG e PostgreSQL 15
apt update
install -d -m 0755 /usr/share/postgresql-common/pgdg
curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc \
  -o /usr/share/postgresql-common/pgdg/apt.postgresql.org.asc
. /etc/os-release
echo "deb [signed-by=/usr/share/postgresql-common/pgdg/apt.postgresql.org.asc] https://apt.postgresql.org/pub/repos/apt ${VERSION_CODENAME}-pgdg main" | \
tee /etc/apt/sources.list.d/pgdg.list
apt update
apt-cache policy postgresql-15 postgresql-client-15
apt install -y postgresql-15 postgresql-client-15 postgresql-contrib-15
systemctl enable --now postgresql
pg_lsclusters
/usr/lib/postgresql/15/bin/psql --version
sudo -u postgres /usr/lib/postgresql/15/bin/psql -p 5432 -c 'SHOW server_version;'
sudo -u postgres /usr/lib/postgresql/15/bin/psql -p 5432 -c 'SHOW data_directory;'
dpkg-query -W postgresql-15 postgresql-client-15


5.4 Hardening iniziale e ascolto rete
cp /etc/postgresql/15/main/postgresql.conf \
  /etc/postgresql/15/main/postgresql.conf.$(date +%F_%H%M).bak
cp /etc/postgresql/15/main/pg_hba.conf \
  /etc/postgresql/15/main/pg_hba.conf.$(date +%F_%H%M).bak
sudo -u postgres psql -c "ALTER SYSTEM SET password_encryption = 'scram-sha-256';"
sudo -u postgres psql -c "ALTER SYSTEM SET listen_addresses = '<IP_POSTGRES>,localhost';"
sudo -u postgres psql -c "ALTER SYSTEM SET log_connections = 'on';"
sudo -u postgres psql -c "ALTER SYSTEM SET log_disconnections = 'on';"
sudo -u postgres psql -c "ALTER SYSTEM SET log_line_prefix = '%m [%p] %q%u@%d ';"
# Sostituire <IP_APP>/32 con IP o subnet reale
printf '
hostssl all all <IP_APP>/32 scram-sha-256
' | sudo tee -a /etc/postgresql/15/main/pg_hba.conf
# Validazione preventiva
sudo -u postgres /usr/lib/postgresql/15/bin/postgres \
  -D /var/lib/postgresql/15/main \
  -c config_file=/etc/postgresql/15/main/postgresql.conf -C config_file
systemctl restart postgresql@15-main
systemctl status postgresql@15-main --no-pager



5.5 Creazione database e utenza applicativa
# Generare la password nel vault; non inserirla nello storico shell.
sudo -u postgres psql <<'SQL'
CREATE ROLE app_owner LOGIN;
CREATE DATABASE appdb OWNER app_owner ENCODING 'UTF8' TEMPLATE template0;
REVOKE CREATE ON SCHEMA public FROM PUBLIC;
\c appdb
ALTER SCHEMA public OWNER TO app_owner;
SQL
# Impostazione password in modo interattivo
sudo -u postgres psql -c "\password app_owner"
# Verifiche
sudo -u postgres psql -c "\du+ app_owner"
sudo -u postgres psql -c "\l+ appdb"


5.6 Parametri iniziali e dimensionamento

5.7 Collaudo PostgreSQL
sudo -u postgres pg_isready -h localhost -p 5432
sudo -u postgres psql -d appdb -c 'SELECT current_database(), current_user, now();'
ss -lntp | grep ':5432'
sudo -u postgres psql -c 'SHOW ssl;'
sudo -u postgres psql -c 'SHOW hba_file;'
sudo -u postgres psql -c 'SELECT * FROM pg_hba_file_rules WHERE error IS NOT NULL;'
sudo journalctl -u postgresql@15-main -n 100 --no-pager


5.8 Backup base e test restore
sudo install -d -o postgres -g postgres -m 0700 /var/backups/postgresql
sudo -u postgres pg_dump -Fc -d appdb \
  -f /var/backups/postgresql/appdb_$(date +%F_%H%M).dump
sudo -u postgres pg_restore --list /var/backups/postgresql/appdb_*.dump | head
# Test restore su DB temporaneo
sudo -u postgres createdb appdb_restore_test
sudo -u postgres pg_restore -d appdb_restore_test \
  /var/backups/postgresql/appdb_*.dump
sudo -u postgres psql -d appdb_restore_test -c '\dt'
sudo -u postgres dropdb appdb_restore_test
